若干行若干数：
```
#include \<iostream\>
#include \<string\>
#include \<sstream\>
#include \<vector\>
using namespace std;

int main(){
    string line;
    while(getline(cin, line)){
        stringstream ss(line);
        vector<int> v; 
        int num;
        while(ss >> num){
            v.push_back(num);
        }
        for(int i = 0; i < v.size(); i++){
            if(i != 0) cout << " ";    //输入空格（末尾无空格）
            cout << v[i];
        }
        cout << endl;
    }
    return 0;
}
```
